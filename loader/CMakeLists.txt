# esprack-loader: the OTA loader of the `loader` partition scheme
# (docs/plans/lib-ota-loader.md). A plain ESP-IDF project, built per chip
# with PlatformIO (platformio.ini) or idf.py.
cmake_minimum_required(VERSION 3.16)

set(PROJECT_VER "0.2.0")  # 0.2: handoff version 2 (v1 still read)
# A dev-key loader says so in its version: the app reads it from the
# factory partition (OtaLoader::devKey(), "devkey") and warns. From the
# generated key header, re-read when it changes.
set(_pubkey "${CMAKE_CURRENT_LIST_DIR}/main/ota_pubkey.h")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_pubkey}")
file(STRINGS "${_pubkey}" _devkey REGEX "^#define ESPRACK_OTA_KEY_DEV +1")
if(_devkey)
  string(APPEND PROJECT_VER "+devkey")
endif()

# Build choices, written as sdkconfig defaults after sdkconfig.defaults and
# its per-chip sdkconfig.defaults.<target> (IDF reads them in that order):
#   -DLOADER_BOOT_PIN=<gpio>    the boot button: the second-stage
#       bootloader's factory-reset GPIO. Held at the level below for
#       LOADER_BOOT_HOLD_S seconds right after a reset, it makes the
#       bootloader erase otadata (the boot choice, no data partition:
#       CONFIG_BOOTLOADER_DATA_FACTORY_RESET is empty) and so boot the
#       factory partition, this loader, even when the app crashes before
#       any of its code runs. Only this project's bootloader has it:
#       tools/flash_loader.ps1 writes that one instead of the app's when
#       the loader's sdkconfig has a pin. -1 (default): none.
#   -DLOADER_BOOT_LEVEL=low|high  the pressed level (default low). The
#       bootloader turns the pin's internal pull-up on: "high" needs an
#       external pull-down of 10 kOhm or less.
#   -DLOADER_BOOT_HOLD_S=<s>   hold time, 1..30, default 5. The bootloader's
#       RTC watchdog (9 s by default, not fed while it waits on the pin,
#       a reset at the end) is set to the hold time + 4 s, as 5 + 4.
#   -DLOADER_FLASH_MODE=qio|dio, -DLOADER_FLASH_FREQ=40m|80m: the flash
#       settings when the board's differ from sdkconfig.defaults.<target>
#       (they must match the app's bootloader: flash_loader.ps1 checks).
# sdkconfig keeps what it already has: a changed choice needs a new
# sdkconfig (build.ps1 makes one each run; PlatformIO: delete
# sdkconfig.<env>).
set(LOADER_BOOT_PIN "-1" CACHE STRING "GPIO that boots the loader when held at reset, -1 none")
set(LOADER_BOOT_LEVEL "low" CACHE STRING "the pressed level: low or high")
set(LOADER_BOOT_HOLD_S "5" CACHE STRING "seconds the pin is held")
set(LOADER_FLASH_MODE "" CACHE STRING "qio or dio, empty: sdkconfig.defaults.<target>")
set(LOADER_FLASH_FREQ "" CACHE STRING "40m or 80m, empty: sdkconfig.defaults.<target>")
if(NOT LOADER_BOOT_PIN MATCHES "^-?[0-9]+$" OR NOT LOADER_BOOT_HOLD_S MATCHES "^[0-9]+$" OR
   LOADER_BOOT_HOLD_S LESS 1 OR LOADER_BOOT_HOLD_S GREATER 30 OR
   NOT LOADER_BOOT_LEVEL MATCHES "^(low|high)$" OR NOT LOADER_FLASH_MODE MATCHES "^(qio|dio)?$" OR
   NOT LOADER_FLASH_FREQ MATCHES "^(40m|80m)?$")
  message(FATAL_ERROR "LOADER_BOOT_PIN / _LEVEL / _HOLD_S or LOADER_FLASH_MODE / _FREQ out of range")
endif()
set(_choices "# generated by CMakeLists.txt from the LOADER_* choices\n")
if(LOADER_BOOT_PIN GREATER_EQUAL 0)
  string(TOUPPER "${LOADER_BOOT_LEVEL}" _level)
  math(EXPR _wdtMs "(${LOADER_BOOT_HOLD_S} + 4) * 1000")
  string(APPEND _choices
    "CONFIG_BOOTLOADER_FACTORY_RESET=y\n"
    "CONFIG_BOOTLOADER_NUM_PIN_FACTORY_RESET=${LOADER_BOOT_PIN}\n"
    "CONFIG_BOOTLOADER_FACTORY_RESET_PIN_${_level}=y\n"
    "CONFIG_BOOTLOADER_OTA_DATA_ERASE=y\n"
    "CONFIG_BOOTLOADER_DATA_FACTORY_RESET=\"\"\n"
    "CONFIG_BOOTLOADER_HOLD_TIME_GPIO=${LOADER_BOOT_HOLD_S}\n"
    "CONFIG_BOOTLOADER_WDT_TIME_MS=${_wdtMs}\n")
else()
  string(APPEND _choices "CONFIG_BOOTLOADER_FACTORY_RESET=n\n")
endif()
if(LOADER_FLASH_MODE)
  string(TOUPPER "${LOADER_FLASH_MODE}" _mode)
  string(APPEND _choices "CONFIG_ESPTOOLPY_FLASHMODE_${_mode}=y\n")
endif()
if(LOADER_FLASH_FREQ)
  string(TOUPPER "${LOADER_FLASH_FREQ}" _freq)
  string(APPEND _choices "CONFIG_ESPTOOLPY_FLASHFREQ_${_freq}=y\n")
endif()
set(_choicesFile "${CMAKE_BINARY_DIR}/loader_choices.defaults")
file(WRITE "${_choicesFile}" "${_choices}")
if(NOT SDKCONFIG_DEFAULTS)
  set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults")
endif()
list(APPEND SDKCONFIG_DEFAULTS "${_choicesFile}")

# Only main and what it needs: no unused components in the image.
set(COMPONENTS main)
# PlatformIO's builder reads its flags from a stub component it adds.
if(EXISTS "$ENV{IDF_PATH}/components/__pio_env")
  list(APPEND COMPONENTS __pio_env)
endif()

include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(esprack-loader)
